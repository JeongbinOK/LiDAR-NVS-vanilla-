  2. utils/graphics.py — 왕복 변환 정확도                                                                                                                                                                                                                  
                                                                                                                                                                                                                                                         
  python3 - <<'EOF'                                                                                                                                                                                                                                        
  import torch                                                                                                                                                                                                                                           
  from utils.misc import points_to_pano  
  from utils.graphics import pano_to_lidar, depth_to_normal
                                                                                                                                                                                                                                                           
  pts = torch.rand(35000, 4) * torch.tensor([60,60,4,1]) + torch.tensor([-30,-30,-2,0])                                                                                                                                                                    
  vfov, hfov = [-30.67, 10.67], [-180, 180]                                                                                                                                                                                                                
                                                                                                                                                                                                                                                           
  ri, mask, _ = points_to_pano(pts, vfov, hfov, 32, 1080)                                                                                                                                                                                                  
  depth = ri[0:1]  # [1, 32, 1080]                                                                                                                                                                                                                         
                                                                                                                                                                                                                                                           
  # 왕복 변환: points -> depth -> points                                                                                                                                                                                                                   
  recovered = pano_to_lidar(depth, vfov, hfov)                                                                                                                                                                                                             
  print(f"Original pts: {pts.shape}, Recovered pts: {recovered.shape}")                                                                                                                                                                                    
  print(f"Depth map coverage: {mask.float().mean()*100:.1f}%")                                                                                                                                                                                           
                                                                                                                                                                                                                                                           
  # Normal map                                                                                                                                                                                                                                             
  normal = depth_to_normal(depth, vfov, hfov)                                                                                                                                                                                                              
  print(f"Normal map shape: {normal.shape}")  # [3, 32, 1080]                                                                                                                                                                                              
  print(f"Normal range: [{normal.min():.3f}, {normal.max():.3f}]")  # [-1, 1]
